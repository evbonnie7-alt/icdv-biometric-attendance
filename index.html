<script>
let attendanceType='IN';
function show(id){
document.querySelectorAll('.step').forEach(x=>x.classList.remove('active'));
document.getElementById(id).classList.add('active');
}
function choose(type){
attendanceType=type;
document.getElementById('selected').textContent=type;
document.getElementById('message').textContent='Place your registered finger';
document.getElementById('finger').textContent='☝';
document.getElementById('finger').classList.remove('scanning');
document.getElementById('scanBtn').disabled=false;
document.getElementById('scanBtn').textContent='Start Fingerprint Scan';
show('scan');
}
function scanFingerprint(){
const btn=document.getElementById('scanBtn');
const finger=document.getElementById('finger');
const message=document.getElementById('message');
btn.disabled=true;
btn.textContent='Scanning...';
finger.classList.add('scanning');
message.textContent='Scanning fingerprint...';
setTimeout(()=>{
finger.classList.remove('scanning');
finger.textContent='✓';
message.textContent='Fingerprint recognized ✓';
setTimeout(()=>{
document.getElementById('result').textContent=
attendanceType+' — Successfully Recorded';
show('success');
},700);
},1600);
}
function restart(){
show('start');
}
